          "Я настільки занепокоєний станом свого тілесного здоров’я, що практично не можу думати ні про що інше."
        ]
      },
      {
        title: "21. Інтерес до статевого життя",
        answers: [
          "Мій інтерес до статевого життя за останній час не зменшився.",
          "Статеве життя мене цікавить зараз менше, ніж звичайно.",
          "Мій інтерес до статевого життя значно зменшився.",
          "Я повністю втратив інтерес до статевого життя."
        ]
      }
    ];

    let currentQuestion = 0;
    let userAnswers = Array(questions.length).fill(null);

    const homeScreen = document.getElementById("homeScreen");
    const testScreen = document.getElementById("testScreen");
    const resultScreen = document.getElementById("resultScreen");
    const questionTitle = document.getElementById("questionTitle");
    const answersBox = document.getElementById("answers");
    const questionCounter = document.getElementById("questionCounter");
    const progressPercent = document.getElementById("progressPercent");
    const progressFill = document.getElementById("progressFill");
    const backBtn = document.getElementById("backBtn");
    const nextBtn = document.getElementById("nextBtn");

    function startTest() {
      homeScreen.style.display = "none";
      resultScreen.style.display = "none";
      testScreen.style.display = "block";
      currentQuestion = 0;
      userAnswers = Array(questions.length).fill(null);
      renderQuestion();
      window.scrollTo({ top: 0, behavior: "smooth" });
    }

    function renderQuestion() {
      const question = questions[currentQuestion];
      questionTitle.textContent = question.title;
      answersBox.innerHTML = "";

      question.answers.forEach((answer, index) => {
        const button = document.createElement("button");
        button.className = "answer-btn";
        button.textContent = answer;
        button.onclick = () => selectAnswer(index);

        if (userAnswers[currentQuestion] === index) {
          button.classList.add("selected");
        }

        answersBox.appendChild(button);
      });

      const answeredCount = userAnswers.filter(answer => answer !== null).length;
      const progress = Math.round((answeredCount / questions.length) * 100);

      questionCounter.textContent = `Питання ${currentQuestion + 1} / ${questions.length}`;
      progressPercent.textContent = `${progress}%`;
      progressFill.style.width = `${progress}%`;

      backBtn.disabled = currentQuestion === 0;
      nextBtn.textContent = currentQuestion === questions.length - 1 ? "Показати результат" : "Далі";
    }

    function selectAnswer(score) {
      userAnswers[currentQuestion] = score;
      renderQuestion();
    }

    function nextQuestion() {
      if (userAnswers[currentQuestion] === null) {
        alert("Оберіть один варіант відповіді, щоб продовжити.");
        return;
      }

      if (currentQuestion < questions.length - 1) {
        currentQuestion++;
        renderQuestion();
        window.scrollTo({ top: 0, behavior: "smooth" });
      } else {
        showResult();
      }
    }

    function prevQuestion() {
      if (currentQuestion > 0) {
        currentQuestion--;
        renderQuestion();
        window.scrollTo({ top: 0, behavior: "smooth" });
      }
    }

    function showResult() {
      const totalScore = userAnswers.reduce((sum, score) => sum + score, 0);
      const result = getResult(totalScore);
      const suicideQuestionScore = userAnswers[8];

      testScreen.style.display = "none";
      resultScreen.style.display = "block";

      document.getElementById("scoreValue").textContent = totalScore;
      document.getElementById("resultTitle").textContent = result.title;
      document.getElementById("resultText").textContent = result.text;

      document.getElementById("riskWarning").style.display = suicideQuestionScore >= 2 ? "block" : "none";
      window.scrollTo({ top: 0, behavior: "smooth" });
    }

    function getResult(score) {
      if (score <= 9) {
        return {
          title: "Відсутність або мінімальна депресія",
          text: "Результат перебуває в межах норми або свідчить про мінімальні прояви депресивних симптомів. Варто й надалі звертати увагу на свій емоційний стан, сон, рівень втоми та загальне самопочуття."
        };
      }

      if (score <= 15) {
        return {
          title: "Субдепресія",
          text: "Результат може свідчити про емоційний дискомфорт, знижений настрій, втому або внутрішнє напруження. Це не є діагнозом, але може бути приводом уважніше поставитися до свого стану."
        };
      }

      if (score <= 19) {
        return {
          title: "Легка депресія",
          text: "Результат може вказувати на наявність вираженіших депресивних симптомів, які вже можуть впливати на повсякденне життя, мотивацію, сон або концентрацію. За можливості варто звернутися до психолога, психотерапевта або сімейного лікаря."
        };
      }

      if (score <= 29) {
        return {
          title: "Помірна депресія",
          text: "Результат може свідчити про суттєвий вплив симптомів на емоційний стан і повсякденне функціонування. Рекомендовано не залишатися з цим наодинці та звернутися по професійну допомогу."
        };
      }

      return {
        title: "Важка депресія",
        text: "Результат може вказувати на значний рівень депресивних симптомів. Важливо звернутися по професійну допомогу якомога швидше, особливо якщо є думки про самопошкодження або втрату сенсу життя."
      };
    }

    function restartTest() {
      resultScreen.style.display = "none";
      homeScreen.style.display = "block";
      userAnswers = Array(questions.length).fill(null);
      currentQuestion = 0;
      window.scrollTo({ top: 0, behavior: "smooth" });
    }
